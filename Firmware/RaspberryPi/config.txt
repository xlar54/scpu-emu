# SCPU-EMU - Raspberry Pi boot configuration
#
# This is the RAD Expansion Unit's own config.txt, with exactly one change:
# kernel= points at kernel8.img instead of kernel_rad.img.
#
# Do not "tidy" it. RAD's bus timings were tuned on a Pi running precisely these
# settings, and several lines that look like generic tuning are load-bearing:
#
#   arm_freq / core_freq / sdram_freq / over_voltage / sdram_schmoo
#       Every constant in SCPU/scpu.cfg is a count of ARM clock cycles, so the
#       clock is part of the timing configuration, not separate from it. The
#       overvolting is what keeps these clocks stable under sustained load.
#
#   gpio=...
#       These set the Pi's pins to safe states *before* any of our code runs.
#       Without them the GPIOs float or default during the seconds the Pi takes
#       to boot, and the RAD can disturb the C64 bus -- or hold the machine in
#       reset -- before the firmware ever initialises them. Note gpio=8=op,dh:
#       GPIO 8 is /RESET, driven high so the C64 is *not* held in reset.
#
#   h264_freq / isp_freq / v3d_freq / temp_soft_limit
#       Keep the unused GPU blocks quiet and the die cool, so the ARM clock does
#       not get pulled around underneath the timings.
#
#   armstub=rad-prefetch.bin
#       Disables L1 data prefetching, so speculative memory stalls cannot land
#       inside a cycle-counted bus window. Required.

arm_freq=1400
core_freq=600
sdram_freq=700
arm_64bit=1
armstub=rad-prefetch.bin
disable_splash=1
hdmi_blanking=1
initial_turbo=2
force_turbo=1

# The only line changed from RAD's original.
kernel=kernel8.img

over_voltage=8
over_voltage_sdram=8
sdram_schmoo=0x02000020
h264_freq=100
isp_freq=100
v3d_freq=100
temp_soft_limit=70
hdmi_force_hotplug=1
gpu_mem=16
disable_audio_dither=1
pwm_sample_bits=11
ignore_lcd=1

# Pin states applied by the GPU firmware before the ARM starts. See above.
gpio=17=ip
gpio=14=ip
gpio=4-5=ip
gpio=18-19=ip
gpio=9=ip
gpio=8=op,dh
gpio=0-3=ip
gpio=10-13=ip
gpio=20-27=ip
